country_event = {
	id = 55555
	title = "EVTNAME55555"
	desc = "EVTDESC55555"
	picture = "Administration"
	is_triggered_only = yes

	option = {
		name = "EVTOPTA55555"
		# Роутер флота — встроен
		random_owned = {
			limit = { is_capital = yes owner = { fleet_doctrine = fleet_mothballing } }
			owner = { country_event = { id = 555564 days = 0 } }
		}
		random_owned = {
			limit = { is_capital = yes owner = { fleet_doctrine = fleet_expansion } }
			owner = { country_event = { id = 555563 days = 0 } }
		}
		random_owned = {
			limit = { is_capital = yes owner = { fleet_doctrine = active_combat_fleet } }
			owner = { country_event = { id = 555562 days = 0 } }
		}
	}
	option = {
		name = "EVTOPTB55555"
		# Роутер колоний — встроен
		random_owned = {
			limit = { is_capital = yes owner = { colonial_politics = non_colonial } }
			owner = { country_event = { id = 555574 days = 0 } }
		}
		random_owned = {
			limit = { is_capital = yes owner = { colonial_politics = inner_colonisation } }
			owner = { country_event = { id = 555573 days = 0 } }
		}
		random_owned = {
			limit = { is_capital = yes owner = { colonial_politics = external_colonisation } }
			owner = { country_event = { id = 555572 days = 0 } }
		}
	}
	option = {
		name = "EVTOPTC55555"
	}
}

# Текущая: active_combat_fleet → показываем mothballing + expansion
country_event = {
	id = 555562
	title = "EVTNAME55556"
	desc = "EVTDESC555562"
	picture = "Administration"
	is_triggered_only = yes

	option = {
		name = "EVTOPTA55556"
		political_reform = fleet_mothballing
		add_country_modifier = { name = fleet_reform_change_cd duration = 3650 }
	}
	option = {
		name = "EVTOPTB55556"
		political_reform = fleet_expansion
		add_country_modifier = { name = fleet_reform_change_cd duration = 3650 }
	}
	option = {
		name = "EVTOPTD55556"
		# Назад — роутер к главному меню
		random_owned = {
			limit = {
				is_capital = yes
				owner = {
					NOT = { has_country_modifier = fleet_reform_change_cd }
					NOT = { has_country_modifier = colonial_reform_change_cd }
				}
			}
			owner = { country_event = { id = 55555 days = 0 } }
		}
		random_owned = {
			limit = {
				is_capital = yes
				owner = {
					NOT = { has_country_modifier = fleet_reform_change_cd }
					has_country_modifier = colonial_reform_change_cd
				}
			}
			owner = { country_event = { id = 555551 days = 0 } }
		}
	}
}
