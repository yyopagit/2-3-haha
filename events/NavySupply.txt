country_event = {
	id = 99510
	title = "noloc"
	desc = "noloc"
	picture = "Administration"
	is_triggered_only = yes

	option = {
		name = "noloc"

		# Склад пополнен — сброс счётчика и предупреждения, штраф НЕ снимаем
		any_country = {
			limit = {
				num_of_ports = 1
				NOT = { fleet_doctrine = fleet_mothballing }
				ammunition = 500
				artillery = 50
				explosives = 500
			}
			clr_country_flag = ns_m1
			clr_country_flag = ns_m2
			clr_country_flag = ns_m3
			clr_country_flag = ns_m4
			remove_country_modifier = naval_supply_shortage_warning
			clr_country_flag = naval_supply_warning_shown
		}

		# Штраф истёк — можно снова получить его в новом цикле дефицита
		any_country = {
			limit = {
				num_of_ports = 1
				NOT = { fleet_doctrine = fleet_mothballing }
				ammunition = 500
				artillery = 50
				explosives = 500
				NOT = { has_country_modifier = underfunded_fleet }
			}
			clr_country_flag = naval_supply_penalty_shown
		}

		# Консервация флота / нет портов — полный сброс
		any_country = {
			limit = {
				OR = {
					fleet_doctrine = fleet_mothballing
					NOT = { num_of_ports = 1 }
				}
			}
			clr_country_flag = ns_m1
			clr_country_flag = ns_m2
			clr_country_flag = ns_m3
			clr_country_flag = ns_m4
			remove_country_modifier = naval_supply_shortage_warning
			remove_country_modifier = underfunded_fleet
			clr_country_flag = naval_supply_warning_shown
			clr_country_flag = naval_supply_penalty_shown
		}
	}
}
