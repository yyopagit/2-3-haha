# Синхронизация кэша штата: state_has_town / state_has_urban
# town_infrastructure = one_per_state — штамп на весь штат от провинции с городом.
# Trigger: один any_owned вместо 4 отдельных сканов. MTTH 7 дней — достаточно.

country_event = {
	id = 910001
	title = "EVTNAME910001"
	desc = "EVTDESC910001"
	picture = "administration"

	trigger = {
		any_owned = {
			OR = {
				AND = {
					has_building = town_infrastructure
					NOT = { has_province_modifier = state_has_town }
				}
				AND = {
					has_province_modifier = state_has_town
					NOT = {
						state_scope = {
							any_owned = { has_building = town_infrastructure }
						}
					}
				}
				AND = {
					terrain = urban
					NOT = { has_province_modifier = state_has_urban }
				}
				AND = {
					has_province_modifier = state_has_urban
					NOT = {
						state_scope = {
							any_owned = { terrain = urban }
						}
					}
				}
			}
		}
	}

	mean_time_to_happen = {
		days = 7
	}

	immediate = {
		any_owned = {
			limit = {
				has_building = town_infrastructure
				NOT = { has_province_modifier = state_has_town }
			}
			state_scope = {
				any_owned = {
					limit = { NOT = { has_province_modifier = state_has_town } }
					add_province_modifier = {
						name = state_has_town
						duration = -1
					}
				}
			}
		}
		any_owned = {
			limit = {
				has_province_modifier = state_has_town
				NOT = {
					state_scope = {
						any_owned = { has_building = town_infrastructure }
					}
				}
			}
			remove_province_modifier = state_has_town
		}
		any_owned = {
			limit = {
				terrain = urban
				NOT = { has_province_modifier = state_has_urban }
			}
			state_scope = {
				any_owned = {
					limit = { NOT = { has_province_modifier = state_has_urban } }
					add_province_modifier = {
						name = state_has_urban
						duration = -1
					}
				}
			}
		}
		any_owned = {
			limit = {
				has_province_modifier = state_has_urban
				NOT = {
					state_scope = {
						any_owned = { terrain = urban }
					}
				}
			}
			remove_province_modifier = state_has_urban
		}
	}

	option = {
		name = "EVTOPTA910001"
		ai_chance = { factor = 100 }
	}
}
