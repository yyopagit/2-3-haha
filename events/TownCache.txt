# Штамп флага is_urban на весь штат, где есть terrain = urban.
# В production_types: has_province_flag (модификатор в тултипе завода крашит).
# Пульс из JAN 140400: догон новых колоний (urban без флага). Без fire_only_once.

country_event = {
	id = 910001
	title = "noloc"
	desc = "noloc"
	is_triggered_only = yes

	immediate = {
		any_country = {
			limit = {
				exists = yes
				any_owned_province = {
					terrain = urban
					NOT = { has_province_flag = is_urban }
				}
			}
			any_owned = {
				limit = {
					terrain = urban
					NOT = { has_province_flag = is_urban }
				}
				state_scope = {
					any_owned = {
						limit = { NOT = { has_province_flag = is_urban } }
						set_province_flag = is_urban
					}
				}
			}
		}
	}

	option = {
		name = "noloc"
		ai_chance = { factor = 100 }
	}
}

# Полный перештамп колониальных модификаторов этой страны.
# Вызов из Reform_change при смене colonial_politics (у реформы allow = always = no,
# ванильного on_reform нет).

	id = 140412
	title = "noloc"
	desc = "noloc"
	is_triggered_only = yes

	immediate = {
		any_owned = {
			limit = { has_province_modifier = external_colonisation_towns }
			remove_province_modifier = external_colonisation_towns
		}
		any_owned = {
			limit = { has_province_modifier = external_colonisation_colonies }
			remove_province_modifier = external_colonisation_colonies
		}
		any_owned = {
			limit = { has_province_modifier = internal_colonisation }
			remove_province_modifier = internal_colonisation
		}

			limit = { has_province_modifier = non_colonial_towns }
			remove_province_modifier = non_colonial_towns
		}
		any_owned = {
			limit = { has_province_modifier = non_colonial_penalty }
			remove_province_modifier = non_colonial_penalty
		}
		any_owned = {
			limit = { has_province_modifier = non_colonial_urban }
			remove_province_modifier = non_colonial_urban
		}
		any_owned = {
			limit = { has_province_modifier = africa_resources }
			remove_province_modifier = africa_resources
		}

			limit = { has_province_modifier = africa_resources_colonial }
			remove_province_modifier = africa_resources_colonial
		}

			limit = {
				has_building = town_infrastructure
				owner = { colonial_politics = external_colonisation }
			}

				name = external_colonisation_towns
				duration = -1
			}
		}
		any_owned = {
			limit = {
				is_overseas = yes
				NOT = { continent = africa }
				owner = { colonial_politics = external_colonisation }
			}
			add_province_modifier = {
				name = external_colonisation_colonies
				duration = -1
			}
		}

			limit = {
				NOT = { has_building = town_infrastructure }
				owner = { colonial_politics = inner_colonisation }
			}
			add_province_modifier = {
				name = internal_colonisation
				duration = -1
			}
		}
		any_owned = {
			limit = {
				has_building = town_infrastructure
				owner = { colonial_politics = non_colonial }
			}
			add_province_modifier = {
				name = non_colonial_towns
				duration = -1
			}
		}
		any_owned = {
			limit = {
				owner = { colonial_politics = non_colonial }
				state_scope = { is_colonial = yes }
			}
			add_province_modifier = {
				name = non_colonial_penalty
				duration = -1
			}
		}
		any_owned = {
			limit = {
				NOT = { has_building = town_infrastructure }

				owner = { colonial_politics = non_colonial }
			}
			add_province_modifier = {
				name = non_colonial_urban
				duration = -1
			}
		}

			limit = {
				continent = africa

			}
			add_province_modifier = {
				name = africa_resources_colonial
				duration = -1
			}
		}
		any_owned = {
			limit = {
				continent = africa
				owner = { NOT = { colonial_politics = external_colonisation } }
			}
			add_province_modifier = {
				name = africa_resources

			}

	option = {
